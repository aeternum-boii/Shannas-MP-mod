country_event = {
	id = aet_ideas.1

	is_triggered_only = yes 
	trigger = {
		has_country_flag = aet_ideas_fiscalCountryRuined
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		clr_country_flag = aet_ideas_fiscalCountryRuined
	}
	
	option = {
		name = aet_ideas.1.a
		ai_chance = { factor = 1 }
		
		add_stability = -3
		add_country_modifier = {
			name = aet_ideas_fiscalCountryRuined
			duration = 3600
		}		
	}
}

country_event = {
	id = aet_ideas.3

			limit = {
				NOT = { has_country_flag = aet_ideas_3_initiated }

				limit = {
					is_colonial_nation = yes 
					NOT = {
						liberty_desire = 75

						value = 1

				log = "[This.GetName]: Ducats to be transfered [This.aet_ideas_ducatsTransfer_a.GetValue]"
			}
			random_subject_country = {
				limit = {
					has_country_flag = aet_ideas_3_eventFlag

						value = modifier:global_tariffs
						who = ROOT
					}
					change_variable = {
						which = aet_temp
						value = 1
					}
					multiply_variable = {
						which = aet_ideas_ducatsTransfer_b
						which = aet_temp
					}				
				}

				ROOT = { set_country_flag = aet_ideas_3_opt2Set }

				log = "[This.GetName]: Ducats to be transfered [This.aet_ideas_ducatsTransfer_b.GetValue]"
			}		
		}
	}


	option = {
		name = aet_ideas.3.a 

		trigger = {
			has_country_flag = aet_ideas_3_opt1Set
		}

		add_country_modifier = {
			name = aet_ideas_recentlyExploitedSubject
			duration = 360
		}

		event_target:aet_ideas_3_a_target = {
			if = {
				limit = {
					has_country_modifier = aet_ideas_exploitedSubject_1
				}
				remove_country_modifier = aet_ideas_exploitedSubject_1
				add_country_modifier = {
					name = aet_ideas_exploitedSubject_2
					duration = 3600
				}
			} else_if = {
				limit = {
					has_country_modifier = aet_ideas_exploitedSubject_0
				}

					name = aet_ideas_exploitedSubject_1

				}		
			} else = {
				add_country_modifier = {
					name = aet_ideas_exploitedSubject_0
					duration = 1800
				}
			}			

			aet_transfer_treasury = {
				taker = PREV

				custom_tooltip = "[Prev.GetName] will receive [This.aet_ideas_ducatsTransfer_a.GetValue] from [This.GetNaem]"
			}
		}

		every_subject_country = {
			limit = {
				has_country_flag = aet_ideas_3_eventFlag
			}
			clr_country_flag = aet_ideas_3_eventFlag
		}

			every_subject_country = {
				limit = {
					is_colonial_nation = yes 
					OR = {
						has_country_modifier = aet_ideas_exploitedSubject_1
						has_country_modifier = aet_ideas_exploitedSubject_2
					}
					liberty_desire = 65

country_event = {
	id = aet_ideas.4

	is_triggered_only = yes 

	trigger = {
		liberty_desire = 65
	}

	option = { # Let us prepare for war immediately 
		name = aet_ideas.4.a

		overlord = {
			country_event = {
				id = aet_ideas.5
				days = 3
			}
		}
		country_event = {
			id = aet_ideas.6
			days = 30
		}

		add_liberty_desire = 15
		create_unit_forcelimit_percentage = {
			type = infantry
			force_limit_percentage = 0.25
		}	
		create_unit_forcelimit_percentage = {
			type = cavalry
			force_limit_percentage = 0.10
		}								
		create_unit_forcelimit_percentage = {
			type = artillery
			force_limit_percentage = 0.05
		}		
		create_unit_forcelimit_percentage = {
			type = heavy_ship
			special_unit_category = man_of_war
			force_limit_percentage = 0.05
		}	
	}

	option = { # Let us bide our time and instigate our fellow colonies 
		name = aet_ideas.4.b

		overlord = {
			country_event = {
				id = aet_ideas.5
				days = 3
			}
		}
		country_event = {
			id = aet_ideas.6
			days = 365
		}

		overlord = {
			every_subject_country = {
				limit = {
					NOT = {
						tag = ROOT
					}
					liberty_desire = 30
					is_colonial_nation = yes 
				}

				random_list = {
					50 = {
						trigger = {
							NOT = {
								liberty_desire = 65 
							}
						}

						modifier = {
							factor = 0.5
							liberty_desire = 40
						}
						modifier = {
							factor = 0.5
							liberty_desire = 50
						}						
					}

					10 = {
						trigger = {
							liberty_desire = 65
						}
						modifier = {
							factor = 3.5
							liberty_desire = 100
						}

						country_event = {
							id = aet_ideas.4
							days = 3
						}
					}

					10 = {
						add_liberty_desire = 5
					}

					20 = {
						add_liberty_desire = 15
					}
				}
			}
		}
	}

	option = { # let us slow down 
		name = aet_ideas.4.c

		add_liberty_desire = -50
		add_stability = -1

		add_country_modifier = {
			name = aet_ideas_gullibleSubject
			duration = 1800
		}

		overlord = {
			every_subject_country = {
				limit = {
					is_colonial_nation = yes 
				}
				add_liberty_desire = -5
			}
		}
	}	
}

			}
		}

		tooltip = {

		}
	}

	option = { # Back down
		name = aet_ideas.6.b

		add_stability = -3 
		add_liberty_desire = -100
		add_country_modifier = {
			name = aet_ideas_gullibleSubject

		}

			limit = {
				alliance_with = PREV 
			}

				who = PREV 
				years = 20
				modifier = break_alliance
				multiplier = 10
			}
		}
	}	
}

country_event = {
	id = aet_ideas.7

	is_triggered_only = yes 

	option = { # Give in to their demands
		name = aet_ideas.7.a
		goto = FROM

		adm_power_cost = 50 
		hidden_effect = { FROM = { grant_independence = yes } }
		create_subject = {
			subject_type = aet_dominion
			subject = FROM 
		}
	}

	option = { # Let them try! 
		name = aet_ideas.7.b
		goto = FROM

		FROM = {
			declare_war_with_cb = {
				casus_belli = cb_colonial_independence_war
				who = ROOT
			}
		}

		every_subject_country = {
			limit = {
				is_colonial_nation = yes 
				liberty_desire = 65
			}

			white_peace = FROM 
			create_alliance = FROM 
			create_unit_forcelimit_percentage = {
				type = infantry
				force_limit_percentage = 0.15
			}	
			join_all_offensive_wars_of = FROM 
		}
	}	
}
