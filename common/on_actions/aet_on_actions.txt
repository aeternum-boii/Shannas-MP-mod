on_startup = { # Make AI actually choose them 
	if = {
		limit = {
			NOT = { is_year = 1445 }
		}
		aet_ideas_removeAllIdeas = yes 
	}
}
on_monthly_pulse = {
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 1
	}
	
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 8
	}
	
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 15
	}
	
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 22
	}
}

on_yearly_pulse = {
	if = {
		limit = {
			has_country_flag = aet_eliteGuards_commonSetupDone
			NOT = { check_variable = {
				which = aet_eliteGuards_actions_bank
				value = 5
			}}
		}

		change_variable = {
			which = aet_eliteGuards_actions_bank
			value = 1 
		}
	}
	if = {
		limit = {
			has_idea_group = grandeur_ideas
		}

		aet_ideas_grandeurApplyModifiers = yes
	}
}

on_five_year_pulse = {
	if = {
		limit = {
			aet_eliteGuards_anyEnabled = yes 
		}
		aet_eliteGuards_refreshAll = yes 
	}
	aet_ideas_refreshModifiers = yes 
}

on_bankruptcy = {
	events = {
		aet_ideas.1
	}
}

on_colonial_liberation = {
		if = {
			limit = {
				has_country_modifier = aet_ideas_exploitedSubject_0
			}
			set_country_flag = aet_ideas_exploitedSubject_0
		} else_if = {
			limit = {
				has_country_modifier = aet_ideas_exploitedSubject_1
			}
			set_country_flag = aet_ideas_exploitedSubject_1
		} else_if = {
			limit = {
				has_country_modifier = aet_ideas_exploitedSubject_2
			}
			set_country_flag = aet_ideas_exploitedSubject_2
		}

		aet_ideas_clearExploitationModifiers = yes 

		# Mainly to keep the buffs in any potential wars 
		if = {
			limit = {
				has_country_flag = aet_ideas_exploitedSubject_0
			}
			clr_country_flag = aet_ideas_exploitedSubject_0
			add_country_modifier = {
				name = aet_ideas_exploitedSubject_0
				duration = 1800
			}
		} else_if = {
			limit = {
				has_country_flag = aet_ideas_exploitedSubject_1
			}
			clr_country_flag = aet_ideas_exploitedSubject_1
			add_country_modifier = {
				name = aet_ideas_exploitedSubject_1
				duration = 1800
			}			
		} else_if = {
			limit = {
				has_country_flag = aet_ideas_exploitedSubject_2
			}
			clr_country_flag = aet_ideas_exploitedSubject_2
			add_country_modifier = {
				name = aet_ideas_exploitedSubject_2
				duration = 1800
			}			
		}
}
