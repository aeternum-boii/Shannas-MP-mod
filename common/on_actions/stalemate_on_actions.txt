on_monthly_pulse = {
    every_country = {
        limit = {
            has_country_flag = enacted_possible_stalemate_1

            NOT = {
                is_in_war = {
                    attacker_leader = event_target:stalemate_1_attacker
                    defender_leader = event_target:stalemate_1_defender
                }
            }
        }

        clr_country_flag = enacted_possible_stalemate_1

        set_variable = {
            which = st_months_passed_1
            value = 0
        }
    }
    every_country = {
        limit = {
            has_country_flag = enacted_possible_stalemate_2
    
            NOT = {
                is_in_war = {
                    attacker_leader = event_target:stalemate_2_attacker
                    defender_leader = event_target:stalemate_2_defender
                }
            }
        }
    
        clr_country_flag = enacted_possible_stalemate_2
    
        set_variable = {
            which = st_months_passed_2
            value = 0
        }
    }
    every_country = {
        limit = {
            has_country_flag = enacted_possible_stalemate_3
    
            NOT = {
                is_in_war = {
                    attacker_leader = event_target:stalemate_3_attacker
                    defender_leader = event_target:stalemate_3_defender
                }
            }
        }
    
        clr_country_flag = enacted_possible_stalemate_3
    
        set_variable = {
            which = st_months_passed_3
            value = 0
        }
    }

    every_country = {
        limit = {
            has_country_flag = enacted_possible_stalemate_1
            check_variable = {
                which = st_months_passed_1
                value = 1
            }
        }

        change_variable = {
            which = st_months_passed_1
            value = -1
        }
    }

    every_country = {
        limit = {
            has_country_flag = enacted_possible_stalemate_2
            check_variable = {
                which = st_months_passed_2
                value = 1
            }
        }

        change_variable = {
            which = st_months_passed_2
            value = -1
        }
    }

    every_country = {
        limit = {
            has_country_flag = enacted_possible_stalemate_3
            check_variable = {
                which = st_months_passed_3
                value = 1
            }
        }

        change_variable = {
            which = st_months_passed_3
            value = -1
        }
    }
}
on_siege_won_country = {
    if = {
        limit = {
            ROOT = {
                is_in_war = {
                    attacker_leader = event_target:stalemate_1_attacker
                    defender_leader = event_target:stalemate_1_defender
                }
            }

            FROM = {
                owner = {
                    war_with = ROOT
                    is_in_war = {
                        attacker_leader = event_target:stalemate_1_attacker
                        defender_leader = event_target:stalemate_1_defender
                        participants = ROOT
                    }
                }
            }
        }

        every_country = {
            limit = {
                has_country_flag = enacted_possible_stalemate_1
            }

            set_variable = {
                which = st_months_passed_1
                value = 36

                    defender_leader = event_target:stalemate_2_defender

                        attacker_leader = event_target:stalemate_2_attacker
                        defender_leader = event_target:stalemate_2_defender
                        participants = ROOT
                    }
                }
            }
        }

        every_country = {
            limit = {
                has_country_flag = enacted_possible_stalemate_2
            }

            set_variable = {
                which = st_months_passed_2
                value = 36
            }
        }
    }
    if = {
        limit = {
            ROOT = {
                is_in_war = {
                    attacker_leader = event_target:stalemate_3_attacker
                    defender_leader = event_target:stalemate_3_defender
                }
            }

            FROM = {
                owner = {
                    war_with = ROOT

                        attacker_leader = event_target:stalemate_3_attacker
                        defender_leader = event_target:stalemate_3_defender

            }
        }

            limit = {
                has_country_flag = enacted_possible_stalemate_3
            }

            set_variable = {
                which = st_months_passed_3
                value = 36

on_war_ended = {
    # Stalemate war 1 ended
    if = {
        limit = {
            OR = {
                AND = {
                    has_country_flag = stalemate_1_attacker_anchor
                    FROM = {
                        has_country_flag = stalemate_1_defender_anchor
                    }
                }

                AND = {
                    has_country_flag = stalemate_1_defender_anchor
                    FROM = {
                        has_country_flag = stalemate_1_attacker_anchor
                    }
                }
            }
        }

        every_country = {
            limit = {
                has_country_flag = enacted_possible_stalemate_1
            }

            clr_country_flag = enacted_possible_stalemate_1
            clr_country_flag = stalemate_1_enforced

            set_variable = {
                which = st_months_passed_1
                value = 0
            }
        }

        every_country = {
            limit = {
                OR = {
                    has_country_flag = stalemate_1_attacker_anchor
                    has_country_flag = stalemate_1_defender_anchor
                }
            }

            clr_country_flag = stalemate_1_attacker_anchor
            clr_country_flag = stalemate_1_defender_anchor
        }

        clear_global_event_target = stalemate_1_attacker
        clear_global_event_target = stalemate_1_defender
    }
    # Stalemate war 2 ended
    if = {
        limit = {
            OR = {
                AND = {
                    has_country_flag = stalemate_2_attacker_anchor
                    FROM = {
                        has_country_flag = stalemate_2_defender_anchor
                    }
                }

                AND = {
                    has_country_flag = stalemate_2_defender_anchor
                    FROM = {
                        has_country_flag = stalemate_2_attacker_anchor
                    }
                }
            }
        }

        every_country = {
            limit = {
                has_country_flag = enacted_possible_stalemate_2
            }

            clr_country_flag = enacted_possible_stalemate_2
            clr_country_flag = stalemate_2_enforced

            set_variable = {
                which = st_months_passed_2
                value = 0
            }
        }

        every_country = {
            limit = {
                OR = {
                    has_country_flag = stalemate_2_attacker_anchor
                    has_country_flag = stalemate_2_defender_anchor
                }
            }

            clr_country_flag = stalemate_2_attacker_anchor
            clr_country_flag = stalemate_2_defender_anchor
        }

        clear_global_event_target = stalemate_2_attacker
        clear_global_event_target = stalemate_2_defender
    }
    # Stalemate war 3 ended
    if = {
        limit = {
            OR = {
                AND = {
                    has_country_flag = stalemate_3_attacker_anchor
                    FROM = {
                        has_country_flag = stalemate_3_defender_anchor
                    }
                }

                AND = {
                    has_country_flag = stalemate_3_defender_anchor
                    FROM = {
                        has_country_flag = stalemate_3_attacker_anchor
                    }
                }
            }
        }

        every_country = {
            limit = {
                has_country_flag = enacted_possible_stalemate_3
            }

            clr_country_flag = enacted_possible_stalemate_3
            clr_country_flag = stalemate_3_enforced

            set_variable = {
                which = st_months_passed_3
                value = 0
            }
        }

        every_country = {
            limit = {
                OR = {
                    has_country_flag = stalemate_3_attacker_anchor
                    has_country_flag = stalemate_3_defender_anchor
                }
            }

            clr_country_flag = stalemate_3_attacker_anchor
            clr_country_flag = stalemate_3_defender_anchor
        }

        clear_global_event_target = stalemate_3_attacker
        clear_global_event_target = stalemate_3_defender
    }
}
