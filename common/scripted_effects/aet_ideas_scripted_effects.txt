# Fires an event for count times in random provinces having a specified flag 
# @param event id of the province event to fire for its owner 
# @param count how often to fire it 
# @param marker modifier that triggers selection 
aet_ideas_fireProvinceEvent = {
	hidden_effect = {
		set_variable = {
			which = aet_index
			value = 0 
		}
		set_variable = {
			which = aet_test 
			value = $count$ 
		}

		if = {
			limit = {
				NOT = {
					check_variable = {
						which = aet_test
						value = 0 
					}
				}
			}
			set_variable = {
				which = aet_test 
				value = 0 			
			}
			log = aet_ideas_fireProvinceEvent_invalidArgument
		}

		while = {
			limit = {
				NOT = {
					check_variable = {
						which = aet_index 
						which = aet_test
					}
				}
			}

			random_province = {
				limit = {
					has_province_modifier = $marker$
				}

				remove_province_modifier = $marker$
				province_event = {
					id = $event$ 
				}
			}

			change_variable = {
				which = aet_index
				value = 1 
			}
		}


		every_province = {
			limit = {
				has_province_modifier = $marker$
			}
			remove_province_modifier = $marker$
		}
	}
}

aet_ideas_exploitationCalculateJointBuildCost = {
	save_event_target_as = aet_province
	owner = {
		save_event_target_as = aet_colony
		overlord = {
			save_event_target_as = aet_investor
		}
	}

	event_target:aet_investor = {
		export_to_variable = {
			which = aet_localCost 
			value = modifier:local_build_cost
			who = event_target:aet_province
		}
		export_to_variable = {
			which = aet_colonyCost 
			value = modifier:build_cost
			who = event_target:aet_colony 
		}
		export_to_variable = {
			which = aet_ownCost 
			value = modifier:build_cost
		}						

		change_variable = {
			which = aet_ownCost
			which = aet_colonyCost
		}
		change_variable = {
			which = aet_ownCost
			which = aet_localCost
		}
		change_variable = {
			which = aet_ownCost
			value = 0.8
		}				
			

		multiply_variable = {
			which = aet_ownCost
			value = $cost$
		}
		multiply_variable = {
			which = aet_ownCost
			value = -1
		}		

		[[compound]change_variable = {
			which = $compound$
			which = aet_ownCost
		}]

		[[increment]change_variable = {
			which = $increment$
			value = 1	
		}]

		[[apply_cost]aet_applyVar_int = {
			left = add_treasury
			var = aet_ownCost
		}]
	}
}
