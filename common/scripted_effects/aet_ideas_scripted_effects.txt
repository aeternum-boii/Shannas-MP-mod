aet_ideas_refreshModifiers = {
	if = {
		limit = {
			has_idea_group = exploitation_ideas 
		}
		aet_ideas_applyBaseExploitationModifier = yes 
	}
}
aet_ideas_clearExploitationModifiers = {
	country_event = {
		id = aet_ideas.2
		days = 1
	}
	remove_country_modifier = aet_ideas_exploitationSubject_0
	remove_country_modifier = aet_ideas_exploitationSubject_1
	remove_country_modifier = aet_ideas_exploitationSubject_2
	remove_country_modifier = aet_ideas_exploitedSubject_0
	remove_country_modifier = aet_ideas_exploitedSubject_1
	remove_country_modifier = aet_ideas_exploitedSubject_2
}
aet_ideas_clearExploitationModifiersProvince = {

}
aet_ideas_applyBaseExploitationModifier = {
	if = {
		limit = {
			exploitation_ideas = 7 
		}

		every_subject_country = {
			limit = {
				is_colonial_nation = yes
				NOT = { has_country_modifier = aet_ideas_exploitationSubject_2 }
			}
			add_country_modifier = {
				name = aet_ideas_exploitationSubject_2
				duration = -1
			}
		}
	} else_if = {
		limit = {
			exploitation_ideas = 6 
		}

		every_subject_country = {
			limit = {
				is_colonial_nation = yes
				NOT = { has_country_modifier = aet_ideas_exploitationSubject_1 }
			}
			add_country_modifier = {
				name = aet_ideas_exploitationSubject_1
				duration = -1
			}
		}
	} else_if = {
		limit = {
			exploitation_ideas = 3 
		}

		every_subject_country = {
			limit = {
				is_colonial_nation = yes
				NOT = { has_country_modifier = aet_ideas_exploitationSubject_0 }
			}
			add_country_modifier = {
				name = aet_ideas_exploitationSubject_0
				duration = -1
			}
		}
	}
}
