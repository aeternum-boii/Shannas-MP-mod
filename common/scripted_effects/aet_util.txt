# Returns the Current Date as used in aet (so 360 day years, 30 day months) 
# @param var the return variable 
aet_getCurrentDate = {
	export_to_variable = {
		variable_name = current_month
		value = trigger_value:is_month
	}
	export_to_variable = {
		variable_name = current_year
		value = trigger_value:is_year
	}

	multiply_variable = {
		which = current_month
		value = 30
	}

	multiply_variable = {
		which = current_year
		value = 360
	}

	change_variable = {
		which = current_month
		which = current_year
	}

	set_variable = {
		which = $var$
		which = current_month
	}
}

# Returns the current date with offset $offset$
# @param var the return variable 
# @param offset the offset in days (raw integer)
aet_getCurrentDateOffset = {
	aet_getCurrentDate = {
		var = $var$
	}
	change_variable = {
		which = $var$
		value = $offset$
	}
}


# Provides a way to dynamically generate event target names with a number prefix 
# @param string the string for the event target
# @param return the return variable 
# @see aet_executeInEventTarget
# @see aet_clearGlobalEventTarget
aet_saveGlobalEventTarget = {
	aet_applyPrefixed = {
		var = aet_util_globalEventTargetCounter
		string = $string$
		left = save_global_event_target_as
	}
	set_variable = {
		which = $return$
		which = aet_util_globalEventTargetCounter
	}

	if = {
		limit = {
			check_variable = {
				which = aet_util_globalEventTargetCounter
				value = 4096
			}
		}
		set_variable = {
			which = aet_util_globalEventTargetCounter
			value = -4097
		}
		log = aet_saveGlobalEventTarget_overflow # In reality harmless, as usually dynamic event targets don't persist that long 
	}

	change_variable = {
		which = aet_util_globalEventTargetCounter
		value = 1
	}
}

# Copy of transfer_treasury without the giver (so FROM = { FROM = {}} bs is not necessary)
#Make sure to make your own custom tooltip for it because parameters can't be given as input. Sadly. 
aet_transfer_treasury = {
	#custom_tooltip = transfer_treasury_tt	#Transfer §Y[$giver$.$variable$.GetValue]§! Ducats to $Y[$taker$.GetName]§!.
	hidden_effect = {
			while = {
				limit = {
					check_variable = {
						which = $variable$
						value = 1
					}
				}
				$taker$ = {
					add_treasury = 1
				}
				add_treasury = -1
				subtract_variable = {
					which = $variable$
					value = 1
				}
			}
			multiply_variable = {
				which = $variable$
				value = 100
			}
			while = {
				limit = {
					check_variable = {
						which = $variable$
						value = 1
					}
				}
				$taker$ = {
					add_treasury = 0.01
				}
				add_treasury = -0.01
				subtract_variable = {
					which = $variable$
					value = 1
				}
			}
	}
	[[custom_tooltip]
		custom_tooltip = $custom_tooltip$
	]
}
