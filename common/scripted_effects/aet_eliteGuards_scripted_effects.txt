aet_eliteGuards_initialSetup = {
	if {
		limit = {
			[[conditional] NOT = {
				has_country_flag = aet_eliteGuards_commonSetupDone
			}]
		}
		hidden_effect = { aet_eliteGuards_init_vars = yes }
		set_country_flag = aet_eliteGuards_commonSetupDone
		add_country_modifier = {
			name = aet_eliteGuardsFL_1
			duration = -1
		}
		set_variable = {
			which = aet_eliteGuards_FL_deFactoLimit
			value = 2
		}
		aet_eliteGuards_refreshAll = yes
	}
}

aet_eliteGuards_refreshAll = {
	aet_eliteGuards_recalcFL = yes
	aet_eliteGuards_checkDecisions = yes 
}

aet_eliteGuards_checkDecisions = {
	if = {
		limit = {
			NOT = { aet_eliteGuard_heavyCavPersistentRequirements = yes }
		}
		remove_country_modifier = aet_eliteGuards_heavyCavalry_specialisation
		remove_country_modifier = aet_eliteGuards_heavyCavalry_specialisation_country
	}
	if = {
		limit = {
			NOT = { aet_eliteGuard_lightCavPersistentRequirements = yes }
		}
		remove_country_modifier = aet_eliteGuards_lightCavalry_specialisation
		remove_country_modifier = aet_eliteGuards_lightCavalry_country
	}
	if = {
		limit = {
			NOT = { aet_eliteGuard_cavArcherPersistentRequirements = yes }
		}
		remove_country_modifier = aet_eliteGuards_horseArcher_country
		remove_country_modifier = aet_eliteGuards_horseArcher_specialisation
		remove_country_modifier = aet_eliteGuards_horseArcher_horde
		remove_country_modifier = aet_eliteGuards_horseArcher_horde_country		
	}
	if = {
		limit = {
			NOT = { aet_eliteGuard_cavArcherNomadPersistentRequirements = yes }
		}
		remove_country_modifier = aet_eliteGuards_horseArcher_horde
		remove_country_modifier = aet_eliteGuards_horseArcher_horde_country	
	}			
}

aet_eliteGuards_takeFullAction = {
	if {
		limit = {
			NOT = { check_variable = {
				which = aet_eliteGuards_actions_bank
				value = 0
			}}
		}

		if {
			limit = {
				check_variable = {
					which = aet_eliteGuards_actions_bank
					value = -1
				}
			}

			mil_power_cost = 20
		} else_if {
			limit = {
				check_variable = {
					which = aet_eliteGuards_actions_bank
					value = -2
				}
			} 

			mil_power_cost = 40
		} else_if {
			limit = {
				check_variable = {
					which = aet_eliteGuards_actions_bank
					value = -3
				}
			} 

			mil_power_cost = 80
		} else_if {
			limit = {
				check_variable = {
					which = aet_eliteGuards_actions_bank
					value = -4
				}
			} 

			mil_power_cost = 160
		} else_if {
			limit = {
				check_variable = {
					which = aet_eliteGuards_actions_bank
					value = -5
				}
			} 

			mil_power_cost = 320
		} else {
			mil_power_cost = 600
		}
	} 

	subtract_variable = {
		which = aet_eliteGuards_actions_bank
		value = 1
	}
}

aet_eliteGuards_takeHalfAction = {
	aet_getEliteGuardActionCost = {
		var = aet_cost
		half = no 
	}
	aet_applyVar = {
		left = mil_power_cost
		var = aet_cost
	}
}

# Returns the military power cost of an elite guard action.*

# @param var the return variable
# @param half whether this is a half action

aet_getEliteGuardActionCost = {
	if {
		limit = {
			check_variable = {
				which = aet_eliteGuards_actions_bank
				value = -1
			}
		}

		set_variable = {
			which = $var$
			value = 20
		}
	} else_if {
		limit = {
			check_variable = {
				which = aet_eliteGuards_actions_bank
				value = -2
			}
		}

		set_variable = {
			which = $var$
			value = 40
		}
	} else_if {
		limit = {
			check_variable = {
				which = aet_eliteGuards_actions_bank
				value = -3
			}
		}

		set_variable = {
			which = $var$
			value = 80
		}
	} else_if {
		limit = {
			check_variable = {
				which = aet_eliteGuards_actions_bank
				value = -4
			}
		}

		set_variable = {
			which = $var$
			value = 160
		}
	} else_if {
		limit = {
			check_variable = {
				which = aet_eliteGuards_actions_bank
				value = -5
			}
		}

		set_variable = {
			which = $var$
			value = 320
		}
	} else {
		set_variable = {
			which = $var$
			value = 600
		}
	}

	[[half] multiply_variable = {
		which = $var$
		value = 0.585786
	}]
}

aet_eliteGuards_recalcFL = {
	hidden_effect = {
		export_to_variable = {
			variable_name = aet_deIureLimit
			value = modifier:aet_eliteGuards_forceLimit
		}

		multiply_variable = {
			which = aet_deIureLimit
			value = 40
		}

		set_variable = {
			which = aet_proportion
			which = aet_eliteGuards_FL_deFactoLimit
		}

		divide_variable = {
			which = aet_proportion
			which = aet_deIureLimit
		}
	}

	if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 0.50
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			50 = yes
		}
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 0.70
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			70 = yes
		}
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 0.80
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			80 = yes
		}
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 1.25
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			100 = yes
		}
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 1.35
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			125 = yes
		}
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 1.50
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			135 = yes
		}
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 1.75
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			150 = yes
		}
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					which = aet_proportion
					value = 2.00
				}
			}
		}
		aet_eliteGuards_applyFLModifier = {
			175 = yes
		}
	}
	else = {
		aet_eliteGuards_applyFLModifier = {
			200 = yes
		}
	}
}
