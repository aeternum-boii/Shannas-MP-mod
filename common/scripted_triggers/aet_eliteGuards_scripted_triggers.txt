			which = aet_guards

		}

		export_to_variable = {
			which = aet_limit
			value = trigger_value:land_forcelimit
		}

		export_to_variable = {
			which = aet_modifier
			value = modifier:allowed_rev_guard_fraction
		}

		multiply_variable = {
			which = aet_limit
			which = aet_modifier
		}

		change_variable = {
			which = aet_limit
			value = 1
		}
		multiply_variable = {
			which = aet_limit
			value = 1.1
		}		

		check_variable = {

			which = aet_limit
		}
	}
}
