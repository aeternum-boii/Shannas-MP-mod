## HRE EVENTS ##

#Increase Austria chances for emperorship when eligible (until ~1600 for good measure)
country_event = {
	id = ae_historical_hre.100
	title = ae_historical_hre.100.t
	desc = ae_historical_hre.100.d
	picture = HRE_eventPicture
	fire_only_once = yes
	trigger = {
		tag = HAB
		NOT = { ae_gamerule_hre_off = yes }
		OR = {
			ae_gamerule_hre_on = yes
			all_countries_including_self = {
				if = {
					limit = { 
						OR = {
							tag = PREV
							tag = BUR
							tag = HAB
							tag = FRA
							any_owned_province = { 
								OR = {
									region = south_german_region
									region = north_german_region
									region = france_region
									region = low_countries_region
									region = italy_region
								}
							}
							any_core_province = { 
								OR = {
									region = south_german_region
									region = north_german_region
									region = france_region
									region = low_countries_region
									region = italy_region
								}
							}
						}
					}
					full_ai = yes
				}
			}
		}
	}
	
	mean_time_to_happen = {
		months = 1
	}	
	
	immediate = {
	}
	
	
	option = { 
		name = ae_historical_hre.100.a 
		add_country_modifier = { 
			name = ae_hab_emperor
			duration = 55000
		}
	}
}

## FORCE HISTORICAL REFORMATION SPREAD ##

#Protestant North Germany
country_event = {
	id = ae_historical_reformation.100
	title = ae_historical_reformation.100.t
	desc = ae_historical_reformation.100.d
	picture = PROTESTANT_GREAT_REFORMER_eventPicture
	
	trigger = {
		NOT = { ae_gamerule_reformation_off = yes }
		OR = {
			ae_gamerule_reformation_on = yes 
			full_ai = yes
		}
		is_religion_enabled = protestant
		OR = { 
			capital_scope = { region = north_german_region }
			tag = PAL
		}
		NOT = { government = theocracy }
		OR = {
			religion = catholic
			AND = {
				NOT = { religion = protestant }
				OR = {
					tag = BRA
					tag = PRU
					tag = SAX
				}
			}
		}
	}
	
	mean_time_to_happen = {
		months = 24
	}	
	
	immediate = {
	}
	
	
	option = { 
		name = ae_historical_reformation.100.a 
		change_religion = protestant
		capital_scope = {
			limit = {
				NOT = { religion = protestant }
			}
			change_religion = protestant
			add_permanent_province_modifier = {
				name = "religious_zeal_at_conv"
				duration = 9000
			}
		}
		random_owned_province = {
			limit = {
				is_capital = no
				NOT = { religion = protestant }
			}
			change_religion = protestant
			add_permanent_province_modifier = {
				name = "religious_zeal_at_conv"
				duration = 9000
			}
		}
	}
}
