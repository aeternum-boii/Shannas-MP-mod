# Applies a variable $var$ to $left$ 
# acts as if $left$ = $var$ (but works) 
# For Negative Floats use aet_applyVar_neg
# For Integers use aet_applyVar_int
# Runs in average 11 checks, max 12 
# @param $var$ the variable to apply
# 			precision of 3 points < 1
#			precision of 2 points < 5
# 			precision of 1 point else
#			0 <= var < 100
# @param $left$ left side of the expression
aet_applyVar = {
	# See Actual Implementation in aet_util.generated.txt

	if = {
		limit = {
			check_variable = {
				which = $var$
				value = 0
			}
		}
		$left$ = 0.001
	}
}

# Executes an effect in an event_target: prefix_string 
# @param var the variable holding the prefix value -4096 < var < 4096
# @param action the scripted effect that is executed 
# @param string the postfix of the event_target 
aet_executeInEventTarget = {
	# See Actual Implementation in aet_util.generated.txt

	if = {
		limit = {
			check_variable = {
				which = $var$
				value = 0
			}
		}
		event_target:4096_$string$ = { $action$ = yes }
	}	
}
