can_colonize_province = {
	condition = {
		tooltip = "colorlessness_ColonyRegionRestriction"
		potential = {
			FROM = {
				is_colonial_nation = yes
			}
		}
		allow = {
			FROM = {
				capital_scope = {
					colonial_region = ROOT
				}
			}
		}
	}
}

		tooltip = "APH_ONLY_EUROPE"
		potential = {

		}
	}
}
