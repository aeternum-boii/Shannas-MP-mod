aet_eliteGuards_anyEnabled = {
	OR = {
		check_variable = {
			which = aet_eliteGuards_inf_enable
			value = 1
		}
		check_variable = {
			which = aet_eliteGuards_cav_enable
			value = 1
		}
		check_variable = {
			which = aet_eliteGuards_arty_enable
			value = 1
		}				
	}
}
aet_eliteGuards_baseDecisionTrigger = {
	AND = {
		aet_eliteGuards_anyEnabled = yes
		NOT = { has_country_flag = aet_eliteGuards_hideDecisions }
	}
}
aet_eliteGuards_upgradeTrigger = {
	aet_eliteGuards_baseDecisionTrigger = yes 
	NOT = {
		has_country_flag = $modifier$
	}	
}
aet_eliteGuards_upgradeTrigger_inf = {
	aet_eliteGuards_upgradeTrigger = {
		modifier = $modifier$
	}	

	hidden_trigger = { 
		OR = {
			aet_is_debug = yes 

			check_variable = {
				which = aet_eliteGuards_inf_enable
				value = 1
			}
		}		
	}
}
aet_eliteGuards_upgradeTrigger_inf_allow = {

		modifier = $modifier$
	}

	aet_applyVar = {

aet_eliteGuards_isAboveFL = {
	variable_arithmetic_trigger = {
		export_to_variable = {
			which = aet_guards
			value = trigger_value:num_of_revolutionary_guard # This works despite static analysis flagging it 
		}

		export_to_variable = {
			which = aet_limit
			value = trigger_value:land_forcelimit
		}

		export_to_variable = {
			which = aet_modifier
			value = modifier:allowed_rev_guard_fraction
		}

		multiply_variable = {
			which = aet_limit
			which = aet_modifier
		}

		change_variable = {
			which = aet_limit
			value = 1
		}
		multiply_variable = {
			which = aet_limit
			value = 1.1
		}		

		check_variable = {
			which = aet_guards
			which = aet_limit
		}
	}
}
